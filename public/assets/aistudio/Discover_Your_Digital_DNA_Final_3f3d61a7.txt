function bfsPath(start,target){
 const q=[start.slice()],came=new Map([[key(start),null]]);while(q.length){const p=q.shift();if(same(p,target)){const path=[];let k=key(p);while(k){const [x,y]=k.split(",").map(Number);path.unshift([x,y]);k=came.get(k)}return path}
  for(const n of neighbors(p)){const k=key(n);if(!came.has(k)){came.set(k,key(p));q.push(n)}}
 }return []
}

 if(c!==state.lastMove){state.metrics.turns++;state.metrics.routeChanges++;}state.lastMove=c;
 const exits=neighbors(next).length;if(exits>=3)state.metrics.branches++;if(exits<=1)state.metrics.deadEnds++;

 return true
}
function moveEnemies(){

  let candidate=null;
  if(enemy.type==="hunter"){const path=bfsPath(enemy.pos,state.player);if(path.length>1)candidate=path[1]}
  else if(enemy.patrol.length){for(let tries=0;tries<enemy.patrol.length;tries++){enemy.patrolIndex=(enemy.patrolIndex+1)%enemy.patrol.length;const p=enemy.patrol[enemy.patrolIndex];if(!wall(...p)&&bfsPath(enemy.pos,p).length){candidate=p;break}}if(candidate){const path=bfsPath(enemy.pos,candidate);candidate=path.length>1?path[1]:null}}
  if(candidate&&!state.enemies.some((other,j)=>j!==index&&same(other.pos,candidate)))enemy.pos=candidate;
 });
}

 return true
}
